;; These functions return strings rather than booleans, which should be more readable for the purpose of repl-based testing but may prove irritating for unit testing.


user=> (check-equal 1 1)
"Test (= 1 1) passed"
user=> (check-equal 1 0)
"Test (= 1 0) failed"
;; This is what it looks like when a function call is incorrect, and causes an error.
user=> (check-equal 1 1 1)
Error

user=> (check-range 2 1 3)
"Test (<= 1 2 3) passed"
user=> (check-range 1 2 3)
"Test (<= 2 1 3) failed"
user=> (check-range 1 1)
Error


user=> (check-precision 1 1.1 0.2)
"Test passed: 1.1 is within 0.2 of 1"
user=> (check-precision 5 1 0.5)
"Test failed: 1 is not within 0.5 of 5"
user=> (check-precision 2)
Wrong number of args (1) passed to: check.fns/check-precision
The function check-precision cannot be called with one argument.
user=> (check-precision "hello")
Error
